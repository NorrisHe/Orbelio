cmake_minimum_required(VERSION 3.25)
project(Orbelio VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
if(WIN32)
    # vcpkg post-link deployment writes DLLs into a shared output directory.
    set_property(GLOBAL APPEND PROPERTY JOB_POOLS orbelio_link=1)
    set(CMAKE_JOB_POOL_LINK orbelio_link)
endif()
option(ORBELIO_ENABLE_TRACY "Enable Tracy instrumentation (adds profiler helper thread)" OFF)
option(ORBELIO_BUILD_VIEWER "Build the Qt/Vulkan application and platform adapters" ON)
option(ORBELIO_BUILD_VULKAN_RENDERER "Build the independent native Vulkan backend" ${ORBELIO_BUILD_VIEWER})

if(ORBELIO_BUILD_VIEWER AND NOT ORBELIO_BUILD_VULKAN_RENDERER)
    message(FATAL_ERROR "ORBELIO_BUILD_VIEWER requires ORBELIO_BUILD_VULKAN_RENDERER")
endif()

if(ORBELIO_BUILD_VIEWER)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 QuickDialogs2)
    qt_standard_project_setup(REQUIRES 6.8)
endif()
if(ORBELIO_BUILD_VULKAN_RENDERER)
    find_package(Vulkan 1.3 REQUIRED)
    find_package(VulkanMemoryAllocator CONFIG REQUIRED)
    find_package(volk CONFIG REQUIRED)
    if(ORBELIO_ENABLE_TRACY)
        find_package(Tracy CONFIG REQUIRED)
    endif()
endif()

add_library(orbelio_options INTERFACE)
target_compile_features(orbelio_options INTERFACE cxx_std_23)
target_compile_definitions(orbelio_options INTERFACE NOMINMAX)
if(MSVC)
    target_compile_options(orbelio_options INTERFACE /W4 /permissive- /utf-8)
else()
    target_compile_options(orbelio_options INTERFACE -Wall -Wextra -Wpedantic)
endif()

add_subdirectory(src)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
